function(add_gitcore_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE gitcore Qt6::Test)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
